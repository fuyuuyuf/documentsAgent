      function renderDeck() {
        if (!latestItems || !document.body.classList.contains('ready')) return;
        if (!latestItems.length) {
          deck.replaceChildren(Object.assign(document.createElement('span'), { className: 'deck-empty', innerHTML: '<span>空</span>' }));
          return;
        }
        const frag = document.createDocumentFragment();
        let dealIndex = 0;
        for (const item of latestItems) {
          const info = describe(item);
          const card = document.createElement('article');
          card.className = 'fcard'; card.tabIndex = 0;
          card.dataset.url = item.url;
          card.setAttribute('role', 'button');
          card.setAttribute('aria-label', `看板：${info.title}`);
          const ring = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
          ring.setAttribute('class', 'ring'); ring.setAttribute('viewBox', '0 0 32 32');
          const dot = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
          dot.setAttribute('cx', '16'); dot.setAttribute('cy', '16'); dot.setAttribute('r', '14');
          ring.append(dot);
          const ext = document.createElement('span'); ext.className = 'ext'; ext.textContent = 'PDF→Map';
          const t = document.createElement('p'); t.className = 't'; t.textContent = info.title;
          const s = document.createElement('p'); s.className = 's';
          s.textContent = `十大分区 · 证据溯源 · ${info.sizeKb} KB\n${info.stamp} 生成`;
          const go = document.createElement('span'); go.className = 'go'; go.textContent = '点击进入概览 ↗';
          card.append(ring, ext, t, s, go);
          if (!seenOutputs.has(item.url) && !reducedMotion) {
            card.classList.add('deal');
            card.style.setProperty('--deal', (dealIndex++ * .16) + 's');  /* 32s-37s：逐份入场 */
          }
          seenOutputs.add(item.url);
          frag.append(card);
        }
        deck.replaceChildren(frag);
      }
